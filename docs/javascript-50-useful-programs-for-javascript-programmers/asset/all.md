function getStrength(p){

if(p.length < 4){

return 0;

}

var level = 0;

if (p.match(/[a-z]/ig)){

level++;

}

if (p.match(/[0-9]/ig)){

level++;

}

if (p.match(/(.[^a-z0-9])/ig)){

level++;

}

if (p.length < 6 && level > 0){

level--;

}

return level

}
