Introduction


Welcome to "50 JavaScript Exercises: Beginner & Intermediate." This book will take you on a trip to improve your understanding and expertise in JavaScript, one of the most frequently used and dynamic programming languages in the world.

JavaScript, which debuted in the mid-1990s as a scripting language for adding interaction to online pages, has evolved dramatically. It's no longer simply a frontend language; it now powers backends, games, mobile apps, and IoT devices. Because of its versatility, as well as the emergence of new web frameworks and the Node.js environment, it has become a critical talent for developers.

The exercises in this book have been deliberately designed to appeal to both newbies dipping their toes into the world of programming and seasoned coders looking to fine-tune their grip on JavaScript's quirks. Each task is designed to improve problem-solving abilities and cement your comprehension of JavaScript's fundamental concepts, from basic variable declarations to the intricacies of asynchronous operations.

Have fun coding!

// 9-1: Fetch data from a provided URL.

function fetchData(url) {

fetch(url)

.then(response => response.json())

.then(data => console.log(data));

}

fetchData('https://api.example.com/data');

// 9-2: Async function that awaits the result of a fetch.

async function asyncFetch(url) {

let response = await fetch(url);

let data = await response.json();

console.log(data);

}

asyncFetch('https://api.example.com/data');

// 9-3: Convert a callback-based function to a promise-based one.

function callbackFunction(data, callback) {

if (data) {

callback(null, "Data processed successfully");

} else {

callback("Error processing data", null);

}

}

function promiseFunction(data) {

return new Promise((resolve, reject) => {

callbackFunction(data, (error, result) => {

if (error) {

reject(error);

} else {

resolve(result);

}

});

});

}

promiseFunction(true).then(console.log).catch(console.error);

// 9-4: Fetch JSON and handle errors gracefully.

function fetchWithGrace(url) {

fetch(url)

.then(response => {

if (!response.ok) {

throw new Error("Network response was not ok");

}

return response.json();

})

.then(data => console.log(data))

.catch(error => console.log(error.message));

}

fetchWithGrace('https://api.example.com/data');

// 9-5: AJAX request using a callback.

function ajaxRequest(url, callback) {

let xhr = new XMLHttpRequest();

xhr.open("GET", url, true);

xhr.onload = function() {

if (xhr.status >= 200 && xhr.status < 400) {

let data = JSON.parse(xhr.responseText);

callback(null, data);

} else {

callback("Error fetching data", null);

}

};

xhr.onerror = function() {

callback("Network error", null);

};

xhr.send();

}

ajaxRequest('https://api.example.com/data', (error, data) => {

if (error) {

console.log(error);

} else {

console.log(data);

}

});
